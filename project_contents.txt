=== App.tsx ===
const App: React.FC = () => {
  const [activeTab, setActiveTab] = useState<Tab>(Tab.Alphabet);

  const renderContent = () => {
    switch (activeTab) {
      case Tab.Alphabet:
        return <Alphabet />;
      case Tab.Catalog:
        return <Catalog />;
      case Tab.Creator:
        return <Creator />;
      default:
        return <Alphabet />;
    }
  };

  const NavButton: React.FC<{ tabName: Tab; icon: React.ReactNode }> = ({ tabName, icon }) => (
    <button
      onClick={() => setActiveTab(tabName)}
      className={`flex-1 sm:flex-initial sm:w-48 flex items-center justify-center gap-2 px-4 py-3 rounded-t-lg font-orbitron text-sm md:text-base transition-all duration-300 border-b-4 ${
        activeTab === tabName
          ? 'bg-cosmic-purple/50 text-stellar-gold border-stellar-gold'
          : 'bg-transparent text-light-nebula border-transparent hover:bg-cosmic-purple/30 hover:text-white'
      }`}
    >
      {icon}
      <span>{tabName}</span>
    </button>
  );

  return (
    <div className="min-h-screen bg-gradient-to-br from-gray-900 via-cosmic-purple to-black text-white font-cinzel">
      <div className="container mx-auto px-4 py-8">
        <header className="text-center mb-8">
          <h1 className="text-5xl md:text-7xl font-cinzel text-stellar-gold" style={{ textShadow: '0 0 12px rgba(252, 211, 77, 0.8)' }}>
            LUMINAR
          </h1>
          <p className="text-lg md:text-xl text-light-nebula font-orbitron mt-2">
            O Oráculo das Ondas
          </p>
        </header>
        
        <nav className="flex flex-col sm:flex-row justify-center border-b border-indigo-500/30 mb-8">
          <NavButton tabName={Tab.Alphabet} icon={<svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" viewBox="0 0 20 20" fill="currentColor"><path d="M5 3a2 2 0 00-2 2v10a2 2 0 002 2h10a2 2 0 002-2V5a2 2 0 00-2-2H5zm0 2h10v10H5V5z"/></svg>} />
          <NavButton tabName={Tab.Catalog} icon={<svg xmlns="http://www.w3.org/2000/svg" className="h-5 w-5" viewBox="0 0 20 20" fill="currentColor"><path fillRule="evenodd" d="M4 2a2 2 0 00-2 2v12a2 2 0 002 2h12a2 2 0 002-2V4a2 2 0 00-2-2H4zm1.5 3a.5.5 0 000 1h6a.5.5 0 000-1h-6zM5 8a.5.5 0 01.5-.5h6a.5.5 0 010 1h-6A.5.5 0 015 8zm0 3a.5.5 0 01.5-.5h6a.5.5 0 010 1h-6A.5.5 0 015 11z" clipRule="evenodd" /></svg>} />
        </nav>

        <main>
          {renderContent()}
        </main>
      </div>
    </div>
  );
};

export default App;
